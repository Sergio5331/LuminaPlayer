cmake_minimum_required(VERSION 3.20)
project(LuminaPlayer VERSION 0.9.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "Utiliza Windows con MSVC o clang-cl y Qt de ABI MSVC.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets Network)
set(MPV_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/mpv" CACHE PATH "SDK de libmpv")
find_path(MPV_INCLUDE_DIR NAMES mpv/client.h HINTS "${MPV_ROOT}" PATH_SUFFIXES include REQUIRED)
find_library(MPV_LIBRARY NAMES mpv libmpv mpv-2 libmpv-2 HINTS "${MPV_ROOT}" PATH_SUFFIXES lib REQUIRED)
find_file(MPV_DLL NAMES libmpv-2.dll mpv-2.dll HINTS "${MPV_ROOT}" PATH_SUFFIXES bin "" REQUIRED)
add_library(MPV::MPV SHARED IMPORTED)
set_target_properties(MPV::MPV PROPERTIES IMPORTED_IMPLIB "${MPV_LIBRARY}"
    IMPORTED_LOCATION "${MPV_DLL}" INTERFACE_INCLUDE_DIRECTORIES "${MPV_INCLUDE_DIR}")
add_library(lumina STATIC src/WindowsComApartment.h src/MpvPlayer.h src/MpvPlayer.cpp
    src/PlayerControls.h src/PlayerControls.cpp src/MainWindow.h src/MainWindow.cpp src/MainWindowPanels.cpp src/UiIcons.h src/UiIcons.cpp src/ThumbnailPreview.h src/ThumbnailPreview.cpp src/ShortcutDefinitions.h src/SingleInstance.h src/SingleInstance.cpp src/MediaFiles.h src/MediaFiles.cpp)
qt_add_resources(lumina branding PREFIX "/" FILES assets/lumina-logo.png)
target_include_directories(lumina PUBLIC src)
target_link_libraries(lumina PUBLIC Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network MPV::MPV ole32 PRIVATE user32)
target_compile_definitions(lumina PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(lumina PUBLIC /W4 /permissive- /utf-8 /Zc:__cplusplus)
qt_add_executable(LuminaPlayer WIN32 src/main.cpp assets/app.rc)
target_link_libraries(LuminaPlayer PRIVATE lumina)
option(LUMINA_DEPLOY_QT "Copiar Qt junto al ejecutable" ON)
if(LUMINA_DEPLOY_QT)
    get_target_property(_qt_qmake Qt6::qmake IMPORTED_LOCATION)
    get_filename_component(_qt_bin "${_qt_qmake}" DIRECTORY)
    find_program(WINDEPLOYQT_EXECUTABLE windeployqt HINTS "${_qt_bin}" REQUIRED)
endif()
function(lumina_deploy target)
    add_custom_command(TARGET ${target} POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${MPV_DLL}" "$<TARGET_FILE_DIR:${target}>" VERBATIM)
    if(LUMINA_DEPLOY_QT)
        add_custom_command(TARGET ${target} POST_BUILD
            COMMAND "${WINDEPLOYQT_EXECUTABLE}" "$<IF:$<CONFIG:Debug>,--debug,--release>"
                --no-translations "$<TARGET_FILE:${target}>" VERBATIM)
    endif()
endfunction()
lumina_deploy(LuminaPlayer)
qt_add_executable(LuminaThumbnail src/ThumbnailWorker.cpp)
target_link_libraries(LuminaThumbnail PRIVATE Qt6::Core MPV::MPV)
target_compile_options(LuminaThumbnail PRIVATE /utf-8)
lumina_deploy(LuminaThumbnail)
add_dependencies(LuminaPlayer LuminaThumbnail)
option(LUMINA_BUILD_TESTS "Pruebas de integración con escritorio Windows" OFF)
if(LUMINA_BUILD_TESTS)
    enable_testing()
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(LuminaTests tests/PlayerTests.cpp)
    add_dependencies(LuminaTests LuminaThumbnail)
    target_link_libraries(LuminaTests PRIVATE lumina Qt6::Test)
    lumina_deploy(LuminaTests)
    add_test(NAME LuminaIntegration COMMAND LuminaTests -o integration-results.txt,txt)
    set_tests_properties(LuminaIntegration PROPERTIES TIMEOUT 90)
endif()
